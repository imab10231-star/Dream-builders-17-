/** * Dream Builders 17 — Backend (Google Apps Script + Google Sheets) * ----------------------------------------------------------------- * Members শিটের কলাম অর্ডার: * পূর্ন নাম | মোবাইল নম্বর | ইমেইল | NID/জন্ম সনদ | ঠিকানা | * শেয়ার সংখ্যা | পেমেন্ট পদ্ধতি | যোগদানের তারিখ | PIN | Status */ function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let result = { success: false, message: 'অজানা request' }; try { const data = JSON.parse(e.postData.contents); const action = data.action; switch (action) { case 'register': result = registerMember(ss, data); break; case 'login': result = loginMember(ss, data); break; case 'loanApply': result = applyLoan(ss, data); break; case 'adminLogin': result = adminLogin(ss, data); break; case 'listMembers': result = listMembers(ss); break; case 'listLoans': result = listLoans(ss); break; case 'updateMemberStatus': result = updateMemberStatus(ss, data); break; case 'updateLoanStatus': result = updateLoanStatus(ss, data); break; default: result = { success: false, message: 'Unknown action' }; } } catch (err) { result = { success: false, message: err.toString() }; } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { return ContentService .createTextOutput(JSON.stringify({ status: 'Dream Builders 17 API is running' })) .setMimeType(ContentService.MimeType.JSON); } /* ---------------- Helpers ---------------- */ function getOrCreateSheet(ss, name, headers) { let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); sheet.setFrozenRows(1); } return sheet; } /* ---------------- Members ---------------- */ // কলাম: 1 পূর্ন নাম, 2 মোবাইল নম্বর, 3 ইমেইল, 4 NID/জন্ম সনদ, // 5 ঠিকানা, 6 শেয়ার সংখ্যা, 7 পেমেন্ট পদ্ধতি, // 8 যোগদানের তারিখ, 9 PIN, 10 Status function membersSheet(ss) { return getOrCreateSheet(ss, 'Members', [ 'পূর্ন নাম', 'মোবাইল নম্বর', 'ইমেইল', 'NID/জন্ম সনদ', 'ঠিকানা', 'শেয়ার সংখ্যা', 'পেমেন্ট পদ্ধতি', 'যোগদানের তারিখ', 'PIN', 'Status' ]); } function registerMember(ss, data) { const sheet = membersSheet(ss); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][1]) === String(data.mobile)) { return { success: false, message: 'এই মোবাইল নম্বর দিয়ে ইতিমধ্যে নিবন্ধন করা হয়েছে।' }; } } sheet.appendRow([ data.name || '', data.mobile || '', data.email || '', data.nid || '', data.address || '', data.shares || 1, data.payment || '', data.joinDate || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'), data.pin || '', 'approved' ]); return { success: true }; } function loginMember(ss, data) { const sheet = membersSheet(ss); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][1]) === String(data.mobile) && String(rows[i][8]) === String(data.pin)) { return { success: true, name: rows[i][0], shares: rows[i][5], status: rows[i][9] }; } } return { success: false, message: 'মোবাইল নম্বর বা পিন সঠিক নয়।' }; } function listMembers(ss) { const sheet = membersSheet(ss); const rows = sheet.getDataRange().getValues(); const members = []; for (let i = 1; i < rows.length; i++) { members.push({ row: i + 1, name: rows[i][0], mobile: rows[i][1], email: rows[i][2], nid: rows[i][3], address: rows[i][4], shares: rows[i][5], payment: rows[i][6], joinDate: rows[i][7], status: rows[i][9] }); } return { success: true, members: members }; } function updateMemberStatus(ss, data) { const sheet = membersSheet(ss); sheet.getRange(data.row, 10).setValue(data.status); // Status column return { success: true }; } /* ---------------- Loans ---------------- */ function loansSheet(ss) { return getOrCreateSheet(ss, 'Loans', ['Timestamp', 'মোবাইল', 'পরিমাণ', 'কারণ', 'Status']); } function applyLoan(ss, data) { const members = membersSheet(ss).getDataRange().getValues(); let found = false; for (let i = 1; i < members.length; i++) { if (String(members[i][1]) === String(data.mobile)) { found = true; break; } } if (!found) { return { success: false, message: 'এই মোবাইল নম্বরে কোনো সদস্য নিবন্ধিত পাওয়া যায়নি।' }; } loansSheet(ss).appendRow([new Date(), data.mobile, data.amount, data.reason, 'pending']); return { success: true }; } function listLoans(ss) { const sheet = loansSheet(ss); const rows = sheet.getDataRange().getValues(); const loans = []; for (let i = 1; i < rows.length; i++) { loans.push({ row: i + 1, mobile: rows[i][1], amount: rows[i][2], reason: rows[i][3], status: rows[i][4] }); } return { success: true, loans: loans }; } function updateLoanStatus(ss, data) { const sheet = loansSheet(ss); sheet.getRange(data.row, 5).setValue(data.status); return { success: true }; } /* ---------------- Admin ---------------- */ function adminSheet(ss) { const sheet = getOrCreateSheet(ss, 'Admin', ['Mobile', 'PIN']); if (sheet.getLastRow() < 2) { // ডিফল্ট অ্যাডমিন — প্রথমবার লগইনের পর Sheet থেকে বদলে নিন sheet.appendRow(['01700000000', '0000']); } return sheet; } function adminLogin(ss, data) { const sheet = adminSheet(ss); const rows = sheet.getDataRange().getValues(); const mobile = rows[1][0]; const pin = rows[1][1]; if (String(mobile) === String(data.mobile) && String(pin) === String(data.pin)) { return { success: true }; } return { success: false, message: 'ভুল মোবাইল নম্বর অথবা পিন।' }; }
